-- ====================================================================
-- MODULE: PET MANAGER (QUẢN LÝ KHÓA/MỞ KHÓA & DUNG HỢP/TIẾN HÓA PET)
-- BẢN QUY CHUẨN: CHỈ TỰ ĐỘNG DUNG HỢP PET RANK B ĐỔ XUỐNG (GRADE <= 3)
-- ====================================================================
return function(Window, Utils)
    local Players = game:GetService("Players")
    local LocalPlayer = Players.LocalPlayer

    -- Biến cấu hình & trạng thái
    local autoFuseEnabled = false
    local skipLockedPets = true
    local skipEquippedPets = true
    local minFuseGrade = 1
    local maxFuseGrade = 3 -- GIỚI HẠN CỨNG: Chỉ tự ghép Pet từ Rank B đổ xuống (Grade 1 -> 3)
    
    -- Biến chống click đúp & chống spam anti-cheat
    local isProcessingLock = false
    local isProcessingFuse = false
    local lastLockTime = 0
    local lastFuseTime = 0
    local COOLDOWN_DELAY = 3 -- Cooldown 3 giây giữa các lần bấm

    -- ==========================================
    -- HÀM TRUY XUẤT NATIVE PET SYSTEM TỪ GAME
    -- ==========================================
    local function GetPathToolEnv()
        local success, env = pcall(function() return getrenv()._G.PathTool end)
        if success and env then return env end
        return nil
    end

    local function GetGamePlayer()
        local env = GetPathToolEnv()
        if env and env.ClientPlayerManager then
            return env.ClientPlayerManager.GetGamePlayer()
        end
        return nil
    end

    local function GetPetSystem()
        local env = GetPathToolEnv()
        if env then
            if env.PetSystem then return env.PetSystem end
            if type(env.Require) == "function" then
                local ok, sys = pcall(function() return env.Require("PetSystem") end)
                if ok and sys then return sys end
            end
        end
        return nil
    end

    local function GetPetMap()
        local player = GetGamePlayer()
        if player and player.pet and player.pet._itemMap then
            return player.pet._itemMap
        end
        return {}
    end

    local function ShowNotify(title, text, duration)
        pcall(function()
            game:GetService("StarterGui"):SetCore("SendNotification", {
                Title = title or "Pet Manager",
                Text = text or "",
                Duration = duration or 3
            })
        end)
    end

    -- ==========================================
    -- 1. TÍNH NĂNG KHÓA / MỞ KHÓA PET (DEBOUNCE SAFE)
    -- ==========================================
    local function SetAllPetLockStatus(targetLockState, filterGrade)
        local now = tick()
        if isProcessingLock then
            ShowNotify("Cảnh Báo", "Đang xử lý khóa/mở khóa Pet, vui lòng đợi!", 2)
            return
        end
        if now - lastLockTime < COOLDOWN_DELAY then
            ShowNotify("Cảnh Báo", string.format("Vui lòng đợi %.1f giây trước khi bấm lại!", COOLDOWN_DELAY - (now - lastLockTime)), 2)
            return
        end

        isProcessingLock = true
        lastLockTime = now

        local PetSystem = GetPetSystem()
        if not PetSystem then
            warn("❌ [PetManager] Không tìm thấy PetSystem!")
            isProcessingLock = false
            return
        end

        local itemMap = GetPetMap()
        local count = 0

        for guid, pet in pairs(itemMap) do
            local isLocked = pet:IsLock()
            local grade = pet:GetGrade() or 1

            if isLocked ~= targetLockState then
                if not filterGrade or grade == filterGrade then
                    local ok = pcall(function()
                        return PetSystem.ClientLockPet(tostring(guid), targetLockState)
                    end)
                    if ok then
                        count = count + 1
                    end
                    -- Delay 0.05s an toàn chống spam anti-cheat rate limit
                    task.wait(0.05)
                end
            end
        end

        local actionName = targetLockState and "Khóa" or "Mở Khóa"
        ShowNotify("Pet Manager", string.format("Đã %s thành công %d Pet!", actionName, count), 3)
        isProcessingLock = false
    end

    -- ==========================================
    -- 2. TÍNH NĂNG DUNG HỢP / TIẾN HÓA PET (RANK B ĐỔ XUỐNG)
    -- ==========================================
    local function ProcessPetFuseOnce(isManualTrigger)
        local now = tick()
        if isProcessingFuse then
            if isManualTrigger then
                ShowNotify("Cảnh Báo", "Đang xử lý Dung Hợp Pet, vui lòng đợi!", 2)
            end
            return 0
        end

        if isManualTrigger and (now - lastFuseTime < COOLDOWN_DELAY) then
            ShowNotify("Cảnh Báo", string.format("Vui lòng đợi %.1f giây trước khi bấm lại!", COOLDOWN_DELAY - (now - lastFuseTime)), 2)
            return 0
        end

        isProcessingFuse = true
        if isManualTrigger then lastFuseTime = now end

        local env = GetPathToolEnv()
        local PetSystem = GetPetSystem()
        if not (env and PetSystem) then
            isProcessingFuse = false
            return 0
        end

        local itemMap = GetPetMap()
        -- Nhóm các pet theo TmplId (loại pet) và Grade (chỉ lọc Grade 1 -> 3, tức Rank B trở xuống)
        local petGroups = {}

        for guid, pet in pairs(itemMap) do
            local isLocked = pet:IsLock()
            local isEquipped = (pet:GetEquipedIndex() ~= nil and pet:GetEquipedIndex() > 0) or pet:IsInTeam()
            local grade = pet:GetGrade() or 1
            local tmplId = pet:GetTmplId()

            local shouldInclude = true
            if skipLockedPets and isLocked then shouldInclude = false end
            if skipEquippedPets and isEquipped then shouldInclude = false end
            -- Giới hạn cứng: Chỉ tự động ghép các Pet từ Grade 1 đến Grade 3 (Rank B trở xuống)
            if grade < minFuseGrade or grade > maxFuseGrade then shouldInclude = false end

            if shouldInclude and tmplId then
                local groupKey = tostring(tmplId) .. "_" .. tostring(grade)
                if not petGroups[groupKey] then
                    petGroups[groupKey] = {}
                end
                table.insert(petGroups[groupKey], {
                    Guid = tostring(guid),
                    PetObj = pet,
                    IsEquipped = isEquipped,
                    IsLocked = isLocked,
                    Grade = grade,
                    Tmpl = pet:GetTmpl()
                })
            end
        end

        local totalFused = 0
        -- Tiến hành dung hợp theo số lượng phôi yêu cầu (CostTmplAmount) của từng loại pet
        for groupKey, list in pairs(petGroups) do
            if #list >= 2 then
                local samplePet = list[1].PetObj
                local tmpl = list[1].Tmpl
                local grade = list[1].Grade
                local evolveInfo = (tmpl and tmpl.Evolve) and tmpl.Evolve[grade]

                if evolveInfo and evolveInfo.CostTmplAmount then
                    local reqFoodCount = evolveInfo.CostTmplAmount
                    local totalNeeded = reqFoodCount + 1

                    -- Tạo bản sao danh sách để nhặt phôi
                    local groupList = { unpack(list) }

                    while #groupList >= totalNeeded do
                        -- Tìm pet chính (Ưu tiên pet đang trang bị nếu có)
                        local mainIndex = 1
                        for i, p in ipairs(groupList) do
                            if p.IsEquipped then
                                mainIndex = i
                                break
                            end
                        end

                        local mainPet = table.remove(groupList, mainIndex)
                        local foodGuids = {}

                        for i = 1, reqFoodCount do
                            local foodPet = table.remove(groupList, 1)
                            if foodPet then
                                table.insert(foodGuids, foodPet.Guid)
                            end
                        end

                        if #foodGuids == reqFoodCount then
                            local ok = pcall(function()
                                if env.ViewUtil and type(env.ViewUtil.DoRequest) == "function" then
                                    return env.ViewUtil.DoRequest(PetSystem.ClientEvolve, mainPet.Guid, foodGuids)
                                else
                                    return PetSystem.ClientEvolve(mainPet.Guid, foodGuids)
                                end
                            end)
                            if ok then
                                totalFused = totalFused + reqFoodCount
                            end
                            -- Delay 0.2s an toàn giữa các lần ghép
                            task.wait(0.2)
                        end
                    end
                end
            end
        end

        isProcessingFuse = false
        return totalFused
    end

    -- ==========================================
    -- GIAO DIỆN UI (RAYFIELD TAB)
    -- ==========================================
    local PetTab = Window:CreateTab(Utils.t("pet_manager") or "Pet Manager", "dog")

    -- SECTION: LOCK & UNLOCK
    PetTab:CreateSection(" " .. (Utils.t("sec_pet_lock") or "🔒 QUẢN LÝ KHÓA PET") .. " ")

    PetTab:CreateButton({
        Name = Utils.t("btn_unlock_all_pets") or "🔓 Mở Khóa Tất Cả Pet (Unlock All)",
        Info = "Mở khóa toàn bộ Pet trong túi để chuẩn bị dung hợp hoặc tái chế.",
        Callback = function()
            task.spawn(function()
                SetAllPetLockStatus(false)
            end)
        end
    })

    PetTab:CreateButton({
        Name = Utils.t("btn_lock_all_pets") or "🔒 Khóa Tất Cả Pet (Lock All)",
        Info = "Khóa toàn bộ Pet trong túi để tránh lỡ tay bán/tái chế.",
        Callback = function()
            task.spawn(function()
                SetAllPetLockStatus(true)
            end)
        end
    })

    -- SECTION: FUSE & EVOLVE
    PetTab:CreateSection(" " .. (Utils.t("sec_pet_fuse") or "🧬 DUNG HỢP PET (RANK B ĐỔ XUỐNG)") .. " ")

    PetTab:CreateToggle({
        Name = Utils.t("auto_fuse_pets") or "Auto Dung Hợp Pet (Auto Fuse)",
        Info = "Tự động quét và ghép các Pet trùng loại từ Rank B đổ xuống (Grade 1 ➔ 3).",
        CurrentValue = false,
        Flag = "AutoPetFuseToggle",
        Callback = function(Value)
            autoFuseEnabled = Value
        end
    })

    PetTab:CreateButton({
        Name = Utils.t("btn_fuse_now") or "⚡ Dung Hợp Tất Cả Pet Trùng Ngay",
        Info = "Thực hiện ghép tất cả các Pet trùng loại từ Rank B đổ xuống trong túi 1 lần lập tức.",
        Callback = function()
            task.spawn(function()
                local fusedCount = ProcessPetFuseOnce(true)
                ShowNotify("Pet Fuse", string.format("Đã dung hợp thành công %d Pet phôi!", fusedCount or 0), 3)
            end)
        end
    })

    PetTab:CreateToggle({
        Name = Utils.t("skip_locked_pets") or "Bỏ Qua Pet Đang Khóa",
        Info = "Không sử dụng Pet đang bị khóa làm phôi dung hợp.",
        CurrentValue = true,
        Flag = "SkipLockedPetsFlag",
        Callback = function(Value)
            skipLockedPets = Value
        end
    })

    PetTab:CreateToggle({
        Name = Utils.t("skip_equipped_pets") or "Bỏ Qua Pet Đang Ra Trận",
        Info = "Không sử dụng Pet đang ra trận/trang bị làm phôi.",
        CurrentValue = true,
        Flag = "SkipEquippedPetsFlag",
        Callback = function(Value)
            skipEquippedPets = Value
        end
    })

    -- VÒNG LẶP AUTO FUSE
    task.spawn(function()
        while task.wait(4) do
            if autoFuseEnabled then
                pcall(function()
                    ProcessPetFuseOnce(false)
                end)
            end
        end
    end)
end
