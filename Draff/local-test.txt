-- BƯỚC 2: HÀM TẢI LOCAL AN TOÀN (SAFE LOAD TỰ ĐỘNG THỬ MULTI-SOURCE)
-- ==========================================
local function SafeLoad(filePath)
    local content = nil

    -- 1. Thử đọc file trực tiếp qua isfile / readfile của Roblox Executor
    local possiblePaths = {
        "d:/scripts/scripts-linh-tinh/Draff/" .. filePath,
        "Draff/" .. filePath
    }
    for _, p in ipairs(possiblePaths) do
        if isfile and isfile(p) then
            local ok, data = pcall(readfile, p)
            if ok and data then
                content = data
                break
            end
        end
    end

    -- 2. Thử đọc qua Localhost Web Server (nếu đang bật python http.server 8000 / 8002)
    if not content then
        for _, port in ipairs({8000, 8002}) do
            local localUrl = "http://127.0.0.1:" .. port .. "/Draff/" .. filePath .. "?t=" .. tostring(os.time())
            local ok, data = pcall(function() return game:HttpGet(localUrl) end)
            if ok and type(data) == "string" and #data > 0 and not string.find(data, "404 File not found") then
                content = data
                break
            end
        end
    end

    -- 3. Thử đọc qua GitHub Raw làm phương án dự phòng 3
    if not content then
        local rawUrl = "https://raw.githubusercontent.com/chaocauminhlason/scripts-linh-tinh/refs/heads/test/pre-config-manager/Draff/" .. filePath .. "?t=" .. tostring(os.time())
        local ok, data = pcall(function() return game:HttpGet(rawUrl) end)
        if ok and type(data) == "string" and #data > 0 and not string.find(data, "404: Not Found") then
            content = data
        end
    end

    if content then
        local func, err = loadstring(content)
        if func then 
            local okExec, resExec = pcall(func)
            if okExec then
                return resExec
            else
                warn("❌ Lỗi Thực Thi tại file local: " .. filePath .. "\n" .. tostring(resExec))
            end
        else 
            warn("❌ Lỗi Cú Pháp tại file local: " .. filePath .. "\n" .. tostring(err)) 
        end
    else
        warn("❌ SafeLoad thất bại cho file: " .. filePath)
    end
    return nil
end
